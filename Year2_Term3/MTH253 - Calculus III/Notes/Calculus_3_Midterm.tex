\noindent \textbf{Question 1 (2 points).} \\
Find the limit, if it exists, or show that the limit does not exist:
\begin{enumerate}
    \item[a)] $\lim_{(x,y)\to(-1,2)} \frac{x^2y^3}{x^2+2y^2}$
    \item[b)] $\lim_{(x,y)\to(0,0)} \frac{x^2y^3}{x^2+2y^2}$
    \item[c)] For which value of $a$ the following function is continuous at $(0,0)$:
    \[
    f(x,y) = \begin{cases} 
    \frac{x^2y^3}{x^2+2y^2} & \text{if } (x,y) \neq (0,0) \\ 
    a & \text{if } (x,y) = (0,0) 
    \end{cases}
    \]
\end{enumerate}

\noindent \textbf{Question 2 (3 points).}
\begin{enumerate}
    \item[a)] If $f(x,y)=x^2\sin(y)+ye^{xy}$, where $x=s+2t$ and $y=st$, use the Chain Rule to find $\frac{\partial f}{\partial s}$ and $\frac{\partial f}{\partial t}$ when $s=0$ and $t=1$.
    \item[b)] If $x^2-3yz^2+xyz-2=0$ then find $\frac{\partial z}{\partial x}$ and $\frac{\partial z}{\partial y}$.
    \item[c)] Find the directional derivative of the function $f(x,y)=e^x\sin(y)$ at the point $(0,\frac{\pi}{3})$ in the direction of the vector $v=\langle-6,8\rangle$.
\end{enumerate}

\noindent \textbf{Question 3 (3 points).} \\
Find the extreme values of the function $f(x,y)=x^2+y^2+4x-4y$ on the disk $x^2+y^2\leq 9$.

\noindent \textbf{Question 4 (2 points).} \\
Find the volume of the solid under the surface $z=1+x^2y^2$ and above the region enclosed by $x=y^2$ and $x=4$.
